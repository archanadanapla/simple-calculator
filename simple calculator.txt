    <script>
        // Global reference for display input element
        const display = document.getElementById('display');

        /**
         * Appends numbers or decimal points to the display input screen
         */
        function appendNumber(input) {
            if (display.value === '0' || display.value === 'Error') {
                display.value = '';
            }
            if (input === '.' && display.value.endsWith('.')) {
                return;
            }
            display.value += input;
        }

        /**
         * Appends mathematical operators safely without duplicates
         */
        function appendOperator(operator) {
            if (display.value === '' || display.value === 'Error') {
                return;
            }
            const lastChar = display.value.slice(-1);
            if (['+', '-', '*', '/', '%'].includes(lastChar)) {
                display.value = display.value.slice(0, -1) + operator;
            } else {
                display.value += operator;
            }
        }

        /**
         * Clears display screen
         */
        function clearDisplay() {
            display.value = '';
        }

        /**
         * Deletes the last character entered
         */
        function deleteLast() {
            if (display.value === 'Error') {
                display.value = '';
            } else {
                display.value = display.value.slice(0, -1);
            }
        }

        /**
         * Evaluates expression and handles errors (e.g. division by zero) gracefully
         */
        function calculateResult() {
            try {
                if (display.value === '') return;

                // Handle explicit division by zero
                if (display.value.includes('/0')) {
                    display.value = 'Error';
                    return;
                }

                // Safely evaluate math expression
                const result = Function('"use strict"; return (' + display.value + ')')();

                if (!isFinite(result) || isNaN(result)) {
                    display.value = 'Error';
                } else {
                    display.value = Number(result.toFixed(8)).toString();
                }
            } catch (error) {
                display.value = 'Error';
            }
        }
    </script>
